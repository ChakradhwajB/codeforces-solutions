import math
t = int(input())
 
for _ in range(t):
    i = 0
    n = int(input())
    raw_seq = [int(x) for x in input().split()]
    
    seq = [raw_seq[0]]
    for j in range(1, n):
        if raw_seq[j] != seq[-1]:
            seq.append(raw_seq[j])
    n = len(seq)
    final_seq = [seq[0]]
 
 
    while i < n - 1:
        if seq[i+1] >= seq[i]:
            while i < n - 1 and seq[i+1] >= seq[i]:
                i += 1
            final_seq.append(seq[i])  
            
        else:
            while i < n - 1 and seq[i+1] <= seq[i]:
                i += 1
            final_seq.append(seq[i])
 
    all_same = len(set(final_seq)) <= 1
    if all_same:
        print(1)
    else:
        print(len(final_seq))
