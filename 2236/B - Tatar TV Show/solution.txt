t = int(input().strip())
 
for i in range(t):
    n,k = map(int, input().split())
    arr = list(input())
    arr = [int(x) for x in arr]
    for i in range(n):
        if arr[i] == 1:
            if i+k < n:
                arr[i+k] = 1-arr[i+k]
                arr[i] = 0
            elif 0 < (i-k) and arr[i-k] == 1:
                arr[i-k] = 1-arr[i-k]
                arr[i] = 0
 
    if all(x == 0 for x in arr):
        print("YES")
    else:
        print("NO")
