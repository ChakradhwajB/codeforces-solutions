import sys
from collections import defaultdict
 
input = sys.stdin.readline

# We don't even need list(), strings can be indexed directly
arr = input().strip() 
unique_target = len(set(arr))
 
frequency_map = defaultdict(int)
 
L = 0
min_window_size = float('inf')
unique_count = 0
 
for R in range(len(arr)):
    
    char_R = arr[R]
    
    
    if frequency_map[char_R] == 0:
        unique_count += 1
        
   
    frequency_map[char_R] += 1
 
    
    while unique_count == unique_target:
        
        min_window_size = min(min_window_size, R - L + 1)
        
       
        char_L = arr[L]
        frequency_map[char_L] -= 1
        
        
        if frequency_map[char_L] == 0:
            unique_count -= 1
