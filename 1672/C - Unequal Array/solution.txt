    
 
 
t = int(input())
 
for _ in range(t):
    n = int(input())
    arr = [int(x) for x in input().split()]
    idx = []
    for i in range(n-1):
        if i == n:
            pass
        else:
 
            if arr[i] == arr[i+1]:
                
                idx.append(i)
    if len(idx) <= 1:
        print(0)
    else:
        val = idx[-1]-idx[0]
 
        if val <= 1:
            print(val)
        else:
            print(val-1)
