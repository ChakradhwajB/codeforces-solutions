import sys
 
class DSU:
    def __init__(self, n):
 
        self.parent = list(range(n + 1))
        self.size = [1] * (n + 1) 
 
    def find(self, i):
 
        root = i
        while self.parent[root] != root:
            root = self.parent[root]
            
        curr = i
        while curr != root:
            nxt = self.parent[curr]
            self.parent[curr] = root
            curr = nxt
            
        return root
 
    def union(self, i, j):
 
        root_i = self.find(i)
        root_j = self.find(j)
 
        if root_i == root_j:
            return False 
 
        if self.size[root_i] < self.size[root_j]:
            root_i, root_j = root_j, root_i 
 
        self.parent[root_j] = root_i   
        self.size[root_i] += self.size[root_j]
        return True  
 
    def connected(self, i, j):
        return self.find(i) == self.find(j)
 
def solve():
    first_line = sys.stdin.readline().split()
    if not first_line:
        return
    n, m = int(first_line[0]), int(first_line[1])
 
    dsu = DSU(n)
 
    for _ in range(m):
        arr = sys.stdin.readline().split()
        if not arr or arr[0] == '0':
            continue 
            
        initial = int(arr[1]) 
        
        for idx in range(2, len(arr)):
            dsu.union(int(arr[idx]), initial)
 
    
    for i in range(1, n + 1):
        root_of_i = dsu.find(i)
        sys.stdout.write(str(dsu.size[root_of_i]) + " ")
    sys.stdout.write("
")
solve()
