#Implement a 2 pointer-technique.
#Let top = 0 and bottom = 0
 
#UPDATING LOGIC:
#Minimum r MUST be abs(max(towers[0] - cities[0],towers[-1]-cities[-1]))
 
#look ahead. If cities[top+1] <= r+towers[bottom], then it works. For loop this and check untill it breaks
#Then let the bottom pointer increase by one. Check if cities[top] 
 
 
 
def rCheck(r, arrTop, arrBot, lenTop, lenBot):
    top=0
   
    for i in range(lenBot):
        while top < lenTop and arrBot[i] - r <= arrTop[top] <= arrBot[i] + r:
            top += 1 
    if top == lenTop:
        return True
    else:
        return False
 
 
 
low = 0
high = 2* 10**9
 
while low<high:
    mid = (low+high)//2
    
    if rCheck(mid, cities, towers, n, q) != False:
        high = mid
 
    else:
        low = mid+1
