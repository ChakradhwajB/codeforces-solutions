#BINARY SEARCH!!!
 
 
# The bounds of your search space.
# High should be an impossibly large valid answer (e.g., 10**18)
 
 
 
 
import sys
input = sys.stdin.readline
 
 
 
t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    arr = [int(x) for x in input().split()]
    arr.sort()
 
 
    
    
 
    #Check if a value mid is a possible mex
    def check(mid):
        have = [False] * (mid+1)
        leftover = []
        for i in arr:
            if i<=mid and not have [i]:
                have[i] = True
            else:
                leftover.append(i)
        
 
        required = {x for x in range(mid+1)}
        needed = []
        for i in required:
            if have[i] == False:
                needed.append(i)
 
 
        if not needed:
            return True
        
        if not leftover:
            return False
 
        if len(leftover) < len(needed):
            return False
 
 
 
        j = 0
        for i in needed:
            val = i*2 + 1
            while j<len(leftover) and leftover[j] < val:
                j+=1
            if j >= len(leftover):
                return False
            j+=1
        return True
 
 
 
 
 
    low = 0
    high = n
    best_ans = 0
    while low <= high:
        mid = (low + high) // 2
        
        if check(mid):
            # If mid works, it's a candidate! Save it.
            best_ans = mid
            # Now try to find an even BIGGER/BETTER answer
            low = mid + 1
            
        else:
            # If mid fails, it's too big. Go smaller.
            high = mid - 1
    print(best_ans+1)
