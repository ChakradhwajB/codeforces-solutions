n,q = map(int, input().split())
arr = [int(x) for x in input().split()]
 
diff = [0]*(n+2)
 
 
for i in range(q):
    L,R = map(int, input().split())
    diff[L] += 1
    diff[R + 1] -= 1
 
final_arr = [0] * (n + 2)
current_val = 0
for i in range(1, n + 1):
    current_val += diff[i]
    final_arr[i] = current_val
 
processed_arr = final_arr[1:-1]
 
processed_arr.sort()
arr.sort()
total = 0
for i in range(n):
    total+= processed_arr[i] * arr[i]
 
print(total)
