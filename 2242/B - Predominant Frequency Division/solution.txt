import sys
input = sys.stdin.readline
 
t = int(input().strip())
for _ in range(t):
    k = int(input().strip())
    arr = [int(x) for x in input().split()]
 
    pref1 = [0] * (k + 1) 
    pref2 = [0] * (k + 1) 
    
    for i in range(k):
        val1 = 1 if arr[i] == 1 else -1
        val2 = 1 if arr[i] in (1, 2) else -1
        
        pref1[i + 1] = pref1[i] + val1
        pref2[i + 1] = pref2[i] + val2
        
  
    suff2 = [-float('inf')] * (k + 1)
    current_max = -float('inf')
 
    for j in range(k - 1, 0, -1):
        current_max = max(current_max, pref2[j])
        suff2[j - 1] = current_max
 
    possible = False
    for i in range(1, k - 1):
        
        if pref1[i] >= 0 and suff2[i] >= pref2[i]:
            possible = True
            break
            
    if possible:
        print("YES")
    else:
        print("NO")
